【三】校验方法

  # 在包目录下执行（Linux / macOS）
  sha256sum -c MANIFEST.sha256

  # Windows PowerShell
  Get-Content MANIFEST.sha256 | ForEach-Object {
    $h,$f = $_ -split '  ',2; if ((Get-FileHash $f -Algorithm SHA256).Hash
      -ne $h.ToUpper()) { "MISMATCH: $f" } }; "done"
